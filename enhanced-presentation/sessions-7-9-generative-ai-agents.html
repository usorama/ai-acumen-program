from typing import Dict, Any, List
from dataclasses import dataclass
import logging

@dataclass
class ToolPermission:
    tool_name: str
    allowed_operations: List[str]
    max_requests_per_hour: int
    requires_approval: bool

class SecureToolExecutor:
    def __init__(self, permissions: Dict[str, ToolPermission]):
        self.permissions = permissions
        self.execution_log = []
        
    def execute_tool(self, tool_name: str, operation: str, params: Dict[str, Any]):
        # Validate permissions
        if not self._validate_permission(tool_name, operation):
            raise PermissionError(f"Operation {operation} not allowed for {tool_name}")
        
        # Rate limiting
        if self._check_rate_limit(tool_name):
            raise RateLimitError(f"Rate limit exceeded for {tool_name}")
        
        # Audit logging
        self._log_execution(tool_name, operation, params)
        
        # Execute with sandboxing
        return self._safe_execute(tool_name, operation, params)
    
    def _validate_permission(self, tool_name: str, operation: str) -> bool:
        permission = self.permissions.get(tool_name)
        return permission and operation in permission.allowed_operations
    
    def _safe_execute(self, tool_name: str, operation: str, params: Dict[str, Any]):
        try:
            # Execute in isolated environment
            result = self._execute_with_timeout(tool_name, operation, params)
            return {"success": True, "result": result}
        except Exception as e:
            logging.error(f"Tool execution failed: {tool_name}.{operation} - {str(e)}")
            return {"success": False, "error": str(e)}

Python - ReAct Agent Framework 
class ReActAgent:
    def __init__(self, llm, tools, max_iterations=5):
        self.llm = llm
        self.tools = tools
        self.max_iterations = max_iterations
        
    def solve(self, task: str) -> str:
        context = f"Task: {task}\n\n"
        
        for i in range(self.max_iterations):
            # Reasoning step
            reasoning_prompt = f"{context}Thought {i+1}: What should I do next?"
            thought = self.llm.generate(reasoning_prompt)
            context += f"Thought {i+1}: {thought}\n"
            
            # Action step
            action_prompt = f"{context}Action {i+1}:"
            action = self.llm.generate(action_prompt)
            context += f"Action {i+1}: {action}\n"
            
            # Execute action
            if action.startswith("Final Answer:"):
                return action.replace("Final Answer:", "").strip()
            
            # Tool execution
            observation = self._execute_action(action)
            context += f"Observation {i+1}: {observation}\n\n"
        
        return "Max iterations reached. Task incomplete."
    
    def _execute_action(self, action: str) -> str:
        # Parse action and execute appropriate tool
        for tool_name, tool_func in self.tools.items():
            if action.startswith(tool_name):
                try:
                    params = self._parse_action_params(action)
                    result = tool_func(**params)
                    return f"Success: {result}"
                except Exception as e:
                    return f"Error: {str(e)}"
        
        return "Unknown action or tool not available."

import asyncio
import logging
from typing import Dict, List, Any, Optional
from dataclasses import dataclass
from datetime import datetime, timedelta

@dataclass
class AgentTask:
    id: str
    description: str
    priority: int
    deadline: Optional[datetime]
    context: Dict[str, Any]

class EnterpriseAgent:
    def __init__(self, 
                 name: str,
                 llm_provider: str,
                 tools: Dict[str, Any],
                 memory_store: Any):
        self.name = name
        self.llm_provider = llm_provider
        self.tools = tools
        self.memory = memory_store
        self.task_queue = []
        self.is_active = False
        
        # Monitoring and logging
        self.metrics = {
            "tasks_completed": 0,
            "tasks_failed": 0,
            "avg_execution_time": 0,
            "tool_usage": {}
        }
        
        logging.basicConfig(level=logging.INFO)
        self.logger = logging.getLogger(f"Agent-{name}")
    
    async def execute_task(self, task: AgentTask) -> Dict[str, Any]:
        """Execute a single task with full error handling and monitoring"""
        start_time = datetime.now()
        
        try:
            self.logger.info(f"Starting task: {task.id} - {task.description}")
            
            # Load relevant context from memory
            context = await self._load_context(task)
            
            # Plan execution steps
            execution_plan = await self._create_execution_plan(task, context)
            
            # Execute plan with monitoring
            result = await self._execute_plan(execution_plan, task)
            
            # Store results in memory
            await self._store_results(task, result)
            
            # Update metrics
            execution_time = (datetime.now() - start_time).total_seconds()
            self._update_metrics("success", execution_time)
            
            self.logger.info(f"Task completed: {task.id}")
            return {"status": "success", "result": result, "execution_time": execution_time}
            
        except Exception as e:
            self.logger.error(f"Task failed: {task.id} - {str(e)}")
            self._update_metrics("failure", (datetime.now() - start_time).total_seconds())
            return {"status": "error", "error": str(e), "task_id": task.id}
    
    async def _execute_plan(self, plan: List[Dict], task: AgentTask) -> Any:
        """Execute a multi-step plan with error recovery"""
        results = []
        
        for step in plan:
            try:
                if step["type"] == "tool_call":
                    result = await self._execute_tool(step["tool"], step["params"])
                elif step["type"] == "llm_call":
                    result = await self._call_llm(step["prompt"], step.get("context", {}))
                elif step["type"] == "decision":
                    result = await self._make_decision(step["options"], step["criteria"])
                
                results.append({"step": step["id"], "result": result})
                
                # Check if we should continue
                if step.get("stop_on_error") and not result.get("success", True):
                    break
                    
            except Exception as e:
                self.logger.warning(f"Step failed: {step['id']} - {str(e)}")
                if step.get("required", True):
                    raise e
                results.append({"step": step["id"], "error": str(e)})
        
        return results
    
    async def _execute_tool(self, tool_name: str, params: Dict[str, Any]) -> Dict[str, Any]:
        """Execute a tool with proper error handling and rate limiting"""
        if tool_name not in self.tools:
            raise ValueError(f"Tool {tool_name} not available")
        
        # Check rate limits
        await self._check_rate_limit(tool_name)
        
        # Execute with timeout
        try:
            result = await asyncio.wait_for(
                self.tools[tool_name].execute(**params),
                timeout=30  # 30 second timeout
            )
            
            # Update tool usage metrics
            self.metrics["tool_usage"][tool_name] = self.metrics["tool_usage"].get(tool_name, 0) + 1
            
            return {"success": True, "result": result}
            
        except asyncio.TimeoutError:
            raise Exception(f"Tool {tool_name} timed out")
        except Exception as e:
            raise Exception(f"Tool {tool_name} failed: {str(e)}")
    
    def get_health_status(self) -> Dict[str, Any]:
        """Return agent health and performance metrics"""
        return {
            "name": self.name,
            "status": "active" if self.is_active else "inactive",
            "queue_size": len(self.task_queue),
            "metrics": self.metrics,
            "last_heartbeat": datetime.now().isoformat()
        }

Python - Agent Communication Framework 
from enum import Enum
from typing import Dict, Any, List, Optional
import asyncio
import json

class MessageType(Enum):
    TASK_REQUEST = "task_request"
    TASK_RESPONSE = "task_response"
    STATUS_UPDATE = "status_update"
    COORDINATION_REQUEST = "coordination_request"
    RESOURCE_CLAIM = "resource_claim"

@dataclass
class AgentMessage:
    sender_id: str
    receiver_id: str
    message_type: MessageType
    payload: Dict[str, Any]
    timestamp: datetime
    correlation_id: Optional[str] = None

class AgentCommunicationHub:
    def __init__(self):
        self.agents = {}
        self.message_queue = asyncio.Queue()
        self.subscriptions = {}
        
    async def register_agent(self, agent_id: str, agent_ref: Any):
        """Register an agent with the communication hub"""
        self.agents[agent_id] = agent_ref
        self.subscriptions[agent_id] = []
        
    async def send_message(self, message: AgentMessage):
        """Send a message between agents"""
        if message.receiver_id in self.agents:
            await self.agents[message.receiver_id].receive_message(message)
        else:
            # Handle broadcast or routing
            await self._route_message(message)
    
    async def broadcast(self, sender_id: str, message_type: MessageType, payload: Dict[str, Any]):
        """Broadcast a message to all agents"""
        for agent_id in self.agents:
            if agent_id != sender_id:
                message = AgentMessage(
                    sender_id=sender_id,
                    receiver_id=agent_id,
                    message_type=message_type,
                    payload=payload,
                    timestamp=datetime.now()
                )
                await self.send_message(message)

class CoordinatedAgent:
    def __init__(self, agent_id: str, capabilities: List[str], communication_hub: AgentCommunicationHub):
        self.agent_id = agent_id
        self.capabilities = capabilities
        self.hub = communication_hub
        self.current_tasks = {}
        self.resource_locks = set()
        
    async def receive_message(self, message: AgentMessage):
        """Handle incoming messages"""
        if message.message_type == MessageType.TASK_REQUEST:
            await self._handle_task_request(message)
        elif message.message_type == MessageType.COORDINATION_REQUEST:
            await self._handle_coordination_request(message)
        elif message.message_type == MessageType.RESOURCE_CLAIM:
            await self._handle_resource_claim(message)
    
    async def request_coordination(self, task_id: str, required_capabilities: List[str]) -> List[str]:
        """Request coordination with other agents for a complex task"""
        coordination_message = AgentMessage(
            sender_id=self.agent_id,
            receiver_id="broadcast",
            message_type=MessageType.COORDINATION_REQUEST,
            payload={
                "task_id": task_id,
                "required_capabilities": required_capabilities,
                "coordinator": self.agent_id
            },
            timestamp=datetime.now()
        )
        
        await self.hub.broadcast(
            self.agent_id, 
            MessageType.COORDINATION_REQUEST, 
            coordination_message.payload
        )
        
        # Wait for responses and form team
        return await self._form_agent_team(task_id, required_capabilities)

class CustomerOnboardingOrchestrator:
    def __init__(self):
        self.agents = {
            "data_validator": DataValidationAgent(),
            "credit_checker": CreditCheckAgent(), 
            "document_processor": DocumentProcessingAgent(),
            "compliance_officer": ComplianceAgent(),
            "account_creator": AccountCreationAgent(),
            "notification_sender": NotificationAgent()
        }
        
        # Define workflow dependencies
        self.workflow = {
            "validate_customer_data": {
                "agent": "data_validator",
                "depends_on": [],
                "parallel": False
            },
            "process_documents": {
                "agent": "document_processor", 
                "depends_on": ["validate_customer_data"],
                "parallel": True
            },
            "check_credit": {
                "agent": "credit_checker",
                "depends_on": ["validate_customer_data"],
                "parallel": True
            },
            "compliance_review": {
                "agent": "compliance_officer",
                "depends_on": ["process_documents", "check_credit"],
                "parallel": False
            },
            "create_account": {
                "agent": "account_creator",
                "depends_on": ["compliance_review"],
                "parallel": False
            },
            "send_welcome": {
                "agent": "notification_sender",
                "depends_on": ["create_account"],
                "parallel": False
            }
        }
    
    async def orchestrate_onboarding(self, customer_data: Dict[str, Any]) -> Dict[str, Any]:
        """Orchestrate the complete customer onboarding process"""
        task_id = f"onboarding_{customer_data['customer_id']}"
        context = {"customer_data": customer_data, "results": {}}
        
        # Execute workflow based on dependencies
        completed_tasks = set()
        
        while len(completed_tasks) < len(self.workflow):
            # Find tasks ready to execute
            ready_tasks = []
            for task_name, task_config in self.workflow.items():
                if task_name not in completed_tasks:
                    dependencies_met = all(dep in completed_tasks for dep in task_config["depends_on"])
                    if dependencies_met:
                        ready_tasks.append((task_name, task_config))
            
            # Execute ready tasks (parallel where possible)
            if ready_tasks:
                if len(ready_tasks) == 1 or not any(task[1]["parallel"] for task in ready_tasks):
                    # Sequential execution
                    for task_name, task_config in ready_tasks:
                        result = await self._execute_task(task_name, task_config, context)
                        context["results"][task_name] = result
                        completed_tasks.add(task_name)
                else:
                    # Parallel execution
                    parallel_tasks = [task for task in ready_tasks if task[1]["parallel"]]
                    sequential_tasks = [task for task in ready_tasks if not task[1]["parallel"]]
                    
                    # Execute parallel tasks
                    if parallel_tasks:
                        parallel_results = await asyncio.gather(*[
                            self._execute_task(task_name, task_config, context)
                            for task_name, task_config in parallel_tasks
                        ])
                        
                        for (task_name, _), result in zip(parallel_tasks, parallel_results):
                            context["results"][task_name] = result
                            completed_tasks.add(task_name)
                    
                    # Execute sequential tasks
                    for task_name, task_config in sequential_tasks:
                        result = await self._execute_task(task_name, task_config, context)
                        context["results"][task_name] = result
                        completed_tasks.add(task_name)
            
            # Check for failures and decide on recovery
            failed_tasks = [task for task, result in context["results"].items() 
                          if result.get("status") == "failed"]
            
            if failed_tasks:
                recovery_action = await self._determine_recovery_action(failed_tasks, context)
                if recovery_action == "abort":
                    return {"status": "failed", "failed_tasks": failed_tasks, "context": context}
                elif recovery_action == "retry":
                    # Remove failed tasks from completed set for retry
                    for task in failed_tasks:
                        completed_tasks.discard(task)
                        del context["results"][task]
        
        return {
            "status": "completed",
            "customer_id": customer_data["customer_id"],
            "account_id": context["results"]["create_account"].get("account_id"),
            "execution_summary": context["results"]
        }
    
    async def _execute_task(self, task_name: str, task_config: Dict, context: Dict) -> Dict[str, Any]:
        """Execute a single task with the designated agent"""
        agent = self.agents[task_config["agent"]]
        
        try:
            # Prepare task-specific context
            task_context = {
                "task_name": task_name,
                "customer_data": context["customer_data"],
                "previous_results": context["results"]
            }
            
            # Execute task
            result = await agent.execute_task(task_context)
            
            return {
                "status": "completed",
                "result": result,
                "execution_time": result.get("execution_time"),
                "agent": task_config["agent"]
            }
            
        except Exception as e:
            return {
                "status": "failed", 
                "error": str(e),
                "agent": task_config["agent"]
            }
